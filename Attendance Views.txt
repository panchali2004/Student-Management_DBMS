--To get the attendance count-- 

CREATE VIEW attendance_count AS 
SELECT RegNo, Course_Code, SUM(Status = 'Present') AS Present_Count,
SUM(Status = 'Absent') AS Absent_Count 
FROM attendance GROUP BY RegNo, Course_Code;

select * from attendance_count;



--To get the attendance percentage--

CREATE VIEW attendance_percentage AS
SELECT RegNo, Course_Code, Present_Count, Absent_Count,
ROUND((Present_Count * 100.0 / (Present_Count + Absent_Count)), 2) 
AS Attendance_Percentage FROM attendance_count;



--To get the attendance eligibility--

CREATE VIEW attendance_eligibility AS
SELECT RegNo, Course_Code, Present_Count, Absent_Count, Attendance_Percentage,
CASE WHEN Attendance_Percentage >= 80 THEN 'Eligible' ELSE 'Not'
END AS Eligibility
FROM attendance_percentage;

--To get the approved attendance count--

CREATE OR REPLACE VIEW approved_medical_count AS 
SELECT RegNo, Course_Code, COUNT(DISTINCT RegNo, Course_Code) AS Approved_Count
FROM medical WHERE Status = 'Approved' GROUP BY RegNo, Course_Code;



--To get the not approved attendance count--

CREATE VIEW not_approved_medical_count AS
SELECT RegNo, Course_Code, COUNT(*) AS Not_approved_yet_Count
FROM medical WHERE Status = 'Pending' OR Status = 'Rejected'
GROUP BY RegNo, Course_Code;



--To get the attendance count(with medical)--

CREATE VIEW medical_attendance_count AS
SELECT a.RegNo, a.Course_Code, a.Present_Count, a.Absent_Count,
IFNULL(m.Approved_Count, 0) AS Medical_Count,
(a.Present_Count + IFNULL(m.Approved_Count, 0)) AS Adjusted_Present,
(a.Absent_Count - IFNULL(m.Approved_Count, 0)) AS Adjusted_Absent
FROM attendance_count a
LEFT JOIN approved_medical_count m
ON a.RegNo = m.RegNo
AND a.Course_Code = m.Course_Code;



--To get the attendance percentage(with medical)--

CREATE VIEW medical_attendance_percentage AS 
SELECT m.RegNo, m.Course_Code, m.Present_Count, m.Absent_Count, m.Medical_Count, m.Adjusted_Present, m.Adjusted_Absent,ROUND((m.Adjusted_Present * 100.0) /(m.Adjusted_Present + m.Adjusted_Absent),2) 
AS Attendance_Percentage
FROM medical_attendance_count m;



--To get the attendance eligibility(with medical)--

CREATE VIEW medical_attendance_eligibility AS
SELECT m.RegNo, m.Course_Code, m.Present_Count, m.Absent_Count, m.Medical_Count, m.Adjusted_Present, m.Adjusted_Absent, m.Attendance_Percentage,
CASE WHEN m.Attendance_Percentage >= 80 THEN 'Eligible' ELSE 'NOT'
END AS Eligibility
FROM medical_attendance_percentage m;
